! ============================================================
! HQ-R1  (Cisco 2911 router) - head office edge router
! Jobs: OSPF with the core switch, DHCP server for Staff and
! Guest, NAT to the Internet, IPsec VPN to the branch, SSH,
! Syslog and SNMP.
! BEFORE pasting: turn on the security licence and reload
! (BUILD_GUIDE.md, step 3), otherwise the crypto lines fail.
! ============================================================
enable
configure terminal
hostname HQ-R1
no ip domain-lookup
ip domain-name corp.local
service password-encryption
service timestamps log datetime msec
enable secret LabEnable2026
username admin secret LabAdmin2026
banner motd #Authorised access only (lab device)#
login block-for 120 attempts 3 within 60
!
! ---------- Interfaces ----------
interface GigabitEthernet0/0
 description Link to HQ-CORE (inside)
 ip address 10.10.255.1 255.255.255.252
 ip nat inside
 no shutdown
!
interface GigabitEthernet0/1
 description Link to ISP (outside)
 ip address 198.51.100.2 255.255.255.252
 ip nat outside
 no shutdown
!
! ---------- Routing ----------
! Default route to the ISP, shared with the core switch through OSPF
ip route 0.0.0.0 0.0.0.0 198.51.100.1
router ospf 1
 router-id 1.1.1.1
 network 10.10.255.0 0.0.0.3 area 0
 default-information originate
!
! ---------- DHCP (requests are relayed here by HQ-CORE) ----------
ip dhcp excluded-address 10.10.10.1 10.10.10.10
ip dhcp excluded-address 10.10.30.1 10.10.30.10
ip dhcp pool STAFF
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 10.10.20.10
ip dhcp pool GUEST
 network 10.10.30.0 255.255.255.0
 default-router 10.10.30.1
 dns-server 203.0.113.10
!
! ---------- NAT ----------
! Line 1: traffic to the branch is NOT translated (it goes through the VPN)
! Line 2: everything else from HQ is translated to 198.51.100.2
access-list 101 deny ip 10.10.0.0 0.0.255.255 192.168.40.0 0.0.0.255
access-list 101 permit ip 10.10.0.0 0.0.255.255 any
ip nat inside source list 101 interface GigabitEthernet0/1 overload
!
! ---------- Site-to-site IPsec VPN to the branch ----------
! Phase 1 (IKE): how the two routers prove who they are
crypto isakmp policy 10
 encryption aes 256
 authentication pre-share
 group 5
crypto isakmp key VpnLabKey2026 address 198.51.100.6
! Phase 2 (IPsec): how the actual traffic is encrypted
crypto ipsec transform-set VPN-SET esp-aes esp-sha-hmac
! Which traffic goes into the tunnel: HQ networks <-> branch LAN
access-list 110 permit ip 10.10.0.0 0.0.255.255 192.168.40.0 0.0.0.255
crypto map VPN-MAP 10 ipsec-isakmp
 set peer 198.51.100.6
 set transform-set VPN-SET
 match address 110
interface GigabitEthernet0/1
 crypto map VPN-MAP
!
! ---------- Management ----------
! SSH only, and only from the Staff VLAN
ip access-list standard MGMT-ONLY
 permit 10.10.10.0 0.0.0.255
line console 0
 login local
 exec-timeout 15 0
line vty 0 4
 login local
 transport input ssh
 access-class MGMT-ONLY in
 exec-timeout 15 0
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
! Logs go to the Syslog service on SRV1; read-only SNMP for monitoring
logging host 10.10.20.10
snmp-server community LabRO ro
!
end
write memory
